**C. Create src/features/person.js:**

```javascript
import { state, elements } from '../lib/state.js';
import { API, TMDBService, RatingsService } from '../services/api.js';
import { navigateTo } from '../lib/navigation.js';
import { showToast } from '../ui/toast.js';

let currentPersonId = null;
let currentCredits = null;
let currentFilter = 'all'; // all | movie | tv | director
let currentSort = 'date'; // date | rating

export function initPersonPage() {
    // Event delegation on the person page container
}

export async function loadPersonPage(personId) {
    currentPersonId = personId;
    const container = document.querySelector('#view-person .person-page');
    // Show loading skeleton
    container.innerHTML = buildPersonSkeleton();

    try {
        // Fetch person details, credits, and external_ids in parallel
        const [person, credits, externalIds] = await Promise.all([
            TMDBService.fetch(`/person/${personId}?language=${state.currentLanguage === 'tr' ? 'tr-TR' : 'en-US'}`),
            TMDBService.fetch(`/person/${personId}/combined_credits?language=${state.currentLanguage === 'tr' ? 'tr-TR' : 'en-US'}`),
            TMDBService.fetch(`/person/${personId}/external_ids`)
        ]);

        currentCredits = credits;

        // Fetch awards via OMDb if IMDB ID available
        let awards = null;
        if (externalIds && externalIds.imdb_id) {
            try {
                const ratingsData = await RatingsService.getAllRatings(externalIds.imdb_id);
                awards = ratingsData?.awards || null;
            } catch { /* awards are non-critical, fail silently */ }
        }

        // Build the page
        container.innerHTML = buildPersonPage(person, credits, awards);

        // Attach event listeners
        attachPersonEvents(person, credits);

        // Load collaborators (async, non-blocking)
        loadCollaborators(personId, credits);
    } catch (error) {
        container.innerHTML = buildPersonError();
    }
}
```

**Person Bio Card (buildPersonBio):**
- Two-column on desktop (photo left, bio right), single-column stacked on mobile
- Photo: 160x240px (2:3), `--radius-md`, poster style, from `person.profile_path` via TMDB image URL
- Name: h2 (28px Crimson Text per existing h2 style)
- Metadata row: birth date formatted with locale, birthplace, known-for department — `.label` (14px/400/`--text-muted`)
- Bio text: `.body-text` (16px/400/`--text-secondary`), max 4 lines with "Show more" button that toggles expanded state
- Back button at top: arrow icon + "Back" that returns to previous detail modal

**Awards Section (buildPersonAwardsHTML):**
- If `awards` string is not null and not "N/A", display it below the bio section
- Render as a subtle info row: Material Symbol `emoji_events` icon (20px, `--text-muted`) + awards text (14px/400/`--text-secondary`)
- Example output: "Won 1 Oscar. 77 wins & 86 nominations." (string comes directly from OMDb Awards field)
- If awards is null or "N/A", do not render the section at all
- Per CONTEXT.md locked decision: "awards/trivia" on person pages — this satisfies the awards portion using free OMDb data

**Filmography Grid (buildFilmographyGrid):**
- Section header: "Filmography" (h4, 20px Crimson Text)
- Filter chips row: All | Movies | TV Shows | As Director
  - Active chip: `--primary` background, `--text-primary` text, `--radius-full`
  - Inactive chip: `--glass-bg` background, `--glass-border` border, `--text-secondary` text, `--radius-full`
  - Minimum height: 44px (touch target), horizontal padding: `--space-md` (16px)
- Sort control: right-aligned — "Newest first" / "By rating" dropdown or toggle, `.label` styling (14px/400)
- Filter logic:
  - All: deduplicated union of cast + crew by TMDB ID
  - Movies: `credits.cast.filter(c => c.media_type === 'movie')`
  - TV Shows: `credits.cast.filter(c => c.media_type === 'tv')`
  - As Director: `credits.crew.filter(c => c.job === 'Director')`
- Sort logic:
  - Newest first: sort by `release_date || first_air_date` descending
  - By rating: sort by `vote_average` descending
- Limit display to top 80 items (performance — prolific actors can have 200+ credits, sort by vote_count desc before limiting)
- Each item: poster card (reuse movie-card pattern from cards.css), showing poster image, title, year, vote_average
- On click: close person page, open detail modal for that item (`openDetail(id, media_type)`)
- Grid: CSS grid, 3 columns on desktop, 2 on tablet, 2 on mobile

**Frequently Works With (buildCollaboratorsSection):**
- Section header: "Frequently Works With" (h4, 20px Crimson Text)
- Algorithm (from RESEARCH.md, with N+1 fix):
  1. Take person's combined_credits cast entries
  2. Sort by vote_count descending, take top 10 titles
  3. For each of those 10 titles, fetch credits via `TMDBService.getCredits(id, media_type)` (10 API calls max)
  4. Count co-occurrence of other person IDs across all fetched credits
  5. Return top 5-8 collaborators with count >= 2
  6. Display as horizontal scroll of person chips
- Person chip: circular avatar (48x48px, `--radius-full`) + name (14px/400/`--text-secondary`) below
- On tap: navigate to that person's page (`loadPersonPage(collaboratorId)`)
- Show "Loading collaborators..." placeholder while computing
- If no collaborators found (all count < 2), hide the section entirely
