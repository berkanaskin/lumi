## Key Requirements (from CONTEXT.md & UI-SPEC.md)

### Authentication
- **Login wall:** Users must login/signup before accessing app (blocks navigation to discover until authenticated)
- **Methods:** Google OAuth (via Firebase) + email/password
- **Onboarding splash:** Cinematic backdrop (blurred poster) with login form overlay
- **Session persistence:** Store auth token in localStorage, auto-login if valid token exists
- **Tagline:** "Discover what to watch. Instantly. Beautifully."

### Watchlist UX
- **Add to Watchlist button:** Available on movie cards (via overlay) and detail page
- **Watchlist state:** Persisted in localStorage + Firestore (when authenticated)
- **Favorites:** Separate from watchlist (users can mark as favorite)
- **Profile page:** Display saved watchlist and favorites

### API Security (CRITICAL)
**Requirement PLAT-05:** All external API calls must route through server-side Edge Functions. No direct API calls from frontend.

- TMDB API: Proxy via /api/tmdb/search, /api/tmdb/details, /api/tmdb/trending
- YouTube API: Proxy via /api/youtube/search
- Gemini AI: Proxy via /api/gemini/recommend
- Edge Functions: Store API keys in Vercel environment variables (process.env.*)
- Frontend: Only calls /api/* endpoints (Vercel Edge Functions), never external APIs

### Internationalization (PLAT-03)
- **Languages:** English (en) and Turkish (tr)
- **Turkish i18n safety:** toLocaleUpperCase('tr-TR') for capital I character handling
- **Completeness:** All UI strings defined in i18n.js

   ```javascript
   import { getAuth, signInWithPopup, GoogleAuthProvider, signInWithEmailAndPassword, createUserWithEmailAndPassword } from 'firebase/auth';

   export function initAuth() {
     const auth = getAuth();
     const googleBtn = document.getElementById('btn-google');
     const emailForm = document.getElementById('email-login-form');

     googleBtn.addEventListener('click', handleGoogleLogin);
     emailForm.addEventListener('submit', handleEmailLogin);

     // Auto-login if user already has valid session
     auth.onAuthStateChanged(user => {
       if (user) {
         hideLoginWall();
         loadUserProfile(user);
       }
     });
   }

   async function handleGoogleLogin() {
     const auth = getAuth();
     const provider = new GoogleAuthProvider();
     try {
       const result = await signInWithPopup(auth, provider);
       const user = result.user;
       state.currentUser = {
         uid: user.uid,
         email: user.email,
         displayName: user.displayName || user.email
       };
       saveStateToFirestore(state.currentUser);
       hideLoginWall();
     } catch (error) {
       showLoginError('Login failed: ' + error.message);
     }
   }

   async function handleEmailLogin(e) {
     e.preventDefault();
     const auth = getAuth();
     const email = document.getElementById('email-input').value;
     const password = document.getElementById('password-input').value;

     try {
       let result;
       // Try sign-in first, fall back to create account
       try {
         result = await signInWithEmailAndPassword(auth, email, password);
       } catch (error) {
         if (error.code === 'auth/user-not-found') {
           result = await createUserWithEmailAndPassword(auth, email, password);
         } else {
           throw error;
         }
       }

       const user = result.user;
       state.currentUser = {
         uid: user.uid,
         email: user.email,
         displayName: user.email
       };
       saveStateToFirestore(state.currentUser);
       hideLoginWall();
     } catch (error) {
       showLoginError('Authentication failed: ' + error.message);
     }
   }

   function hideLoginWall() {
     const wall = document.getElementById('login-wall');
     wall.classList.remove('active');
     wall.style.pointerEvents = 'none';
   }

   function showLoginError(message) {
     const errorDiv = document.getElementById('login-error');
     errorDiv.textContent = message;
     errorDiv.style.display = 'block';
     setTimeout(() => { errorDiv.style.display = 'none'; }, 5000);
   }

   export function saveToWatchlist(movieId) {
     if (!state.currentUser) {
       showToast('Please log in to save movies');
       return;
     }
     if (!state.watchlist) state.watchlist = [];
     if (!state.watchlist.includes(movieId)) {
       state.watchlist.push(movieId);
       saveStateToFirestore(state);
     }
   }

   function saveStateToFirestore(data) {
     // Save user state to Firestore for persistence across devices
     if (state.currentUser && state.currentUser.uid) {
       const db = getFirestore();
       setDoc(doc(db, 'users', state.currentUser.uid), {
         email: state.currentUser.email,
         displayName: state.currentUser.displayName,
         watchlist: state.watchlist || [],
         favorites: state.favorites || [],
         updatedAt: new Date()
       });
     }
   }
   ```

   ```javascript
   export async function initProfilePage() {
     const logoutBtn = document.getElementById('btn-logout');
     logoutBtn.addEventListener('click', handleLogout);

     // Load and display user profile when page becomes visible
     const profileSection = document.getElementById('profile');
     if (profileSection) {
       const observer = new MutationObserver(() => {
         if (!profileSection.classList.contains('hidden')) {
           loadProfilePage();
         }
       });
       observer.observe(profileSection, { attributes: true, attributeFilter: ['class'] });
     }
   }

   async function loadProfilePage() {
     if (!state.currentUser) return;

     // Display user info
     document.getElementById('user-name').textContent = state.currentUser.displayName || 'Anonymous';
     document.getElementById('user-email').textContent = state.currentUser.email || '';

     // Load watchlist movies
     const watchlistContainer = document.getElementById('watchlist-container');
     const watchlistEmpty = document.getElementById('watchlist-empty');
     watchlistContainer.innerHTML = '';

     if (state.watchlist && state.watchlist.length > 0) {
       watchlistEmpty.style.display = 'none';
       for (const movieId of state.watchlist) {
         const movie = await fetchMovieDetails(movieId); // Fetch from cache/API
         if (movie) {
           const card = createMovieCard(movie);
           card.querySelector('.btn-add-watchlist').textContent = 'Remove from Watchlist';
           card.querySelector('.btn-add-watchlist').addEventListener('click', (e) => {
             e.preventDefault();
             removeFromWatchlist(movieId);
             card.remove();
           });
           watchlistContainer.appendChild(card);
         }
       }
     } else {
       watchlistEmpty.style.display = 'block';
     }

     // Load favorites (similar logic)
     const favoritesContainer = document.getElementById('favorites-container');
     const favoritesEmpty = document.getElementById('favorites-empty');
     favoritesContainer.innerHTML = '';

     if (state.favorites && state.favorites.length > 0) {
       favoritesEmpty.style.display = 'none';
       for (const movieId of state.favorites) {
         const movie = await fetchMovieDetails(movieId);
         if (movie) {
           const card = createMovieCard(movie);
           card.querySelector('.btn-add-watchlist').textContent = 'Remove from Favorites';
           card.querySelector('.btn-add-watchlist').addEventListener('click', (e) => {
             e.preventDefault();
             removeFromFavorites(movieId);
             card.remove();
           });
           favoritesContainer.appendChild(card);
         }
       }
     } else {
       favoritesEmpty.style.display = 'block';
     }
   }

   async function handleLogout() {
     const auth = getAuth();
     await auth.signOut();
     state.currentUser = null;
     state.watchlist = [];
     state.favorites = [];
     saveState();
     showLoginWall();
     navigateTo('discover');
   }

   export function removeFromWatchlist(movieId) {
     if (state.watchlist) {
       state.watchlist = state.watchlist.filter(id => id !== movieId);
       saveState();
       saveStateToFirestore(state);
     }
   }

   export function removeFromFavorites(movieId) {
     if (state.favorites) {
       state.favorites = state.favorites.filter(id => id !== movieId);
       saveState();
       saveStateToFirestore(state);
     }
   }

   export function addToFavorites(movieId) {
     if (!state.currentUser) {
       showToast('Please log in to add favorites');
       return;
     }
     if (!state.favorites) state.favorites = [];
     if (!state.favorites.includes(movieId)) {
       state.favorites.push(movieId);
       saveState();
       saveStateToFirestore(state);
       showToast('Added to favorites');
     }
   }
   ```

   ```javascript
   // api/tmdb.js — Vercel Edge Function
   export default async function handler(request) {
     const { searchParams } = new URL(request.url);
     const endpoint = searchParams.get('endpoint'); // 'search', 'details', 'trending'
     const query = searchParams.get('q');
     const movieId = searchParams.get('id');
     const language = searchParams.get('language') || 'en-US';

     const apiKey = process.env.TMDB_API_KEY; // Stored in Vercel secrets
     if (!apiKey) {
       return new Response(JSON.stringify({ error: 'API key not configured' }), { status: 500 });
     }

     let tmdbUrl;
     if (endpoint === 'search') {
       tmdbUrl = `https://api.themoviedb.org/3/search/multi?query=${encodeURIComponent(query)}&language=${language}&api_key=${apiKey}`;
     } else if (endpoint === 'details') {
       tmdbUrl = `https://api.themoviedb.org/3/movie/${movieId}?language=${language}&api_key=${apiKey}`;
     } else if (endpoint === 'trending') {
       tmdbUrl = `https://api.themoviedb.org/3/trending/movie/week?language=${language}&api_key=${apiKey}`;
     } else {
       return new Response(JSON.stringify({ error: 'Unknown endpoint' }), { status: 400 });
     }

     try {
       const response = await fetch(tmdbUrl);
       if (!response.ok) {
         throw new Error(`TMDB error: ${response.status}`);
       }
       const data = await response.json();
       return new Response(JSON.stringify(data), {
         headers: { 'Content-Type': 'application/json' },
       });
     } catch (error) {
       console.error('[TMDB Proxy] Error:', error);
       return new Response(JSON.stringify({ error: 'Search failed' }), { status: 500 });
     }
   }
   ```
