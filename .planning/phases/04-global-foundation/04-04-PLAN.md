<objective>
Greet first-launch users with a 3-step full-screen wizard: (1) language confirm, (2) country confirm, (3) owned streaming platforms multi-select. Pre-populate from `locale.js` defaults. Persist incrementally to localStorage; for authenticated users, also sync to Firestore `users/{uid}.preferences`. Skip allowed on all steps with sensible defaults.

Purpose: AI recommendations and region-specific UX (streaming providers, broadcast info) need a minimum profile: lang + country + owned platforms. Onboarding captures this in <30 seconds without forcing accounts (works for guests; auth optional post-onboarding).

Output: New `src/features/onboarding.js` wizard module, full-screen CSS, boot-time trigger in `src/main.js`, TMDB `/watch/providers/tv` proxy support in `api/tmdb.js`, and unit + integration tests.
</objective>

       ```js
       import { getLocale, setLocale, COUNTRY_TO_LANG } from '../lib/locale.js';

       export const COUNTRY_SHORTLIST = ['TR','US','GB','DE','FR','ES','IT','NL','BE','SE','NO','DK','FI','PL','RU','JP','KR','CN','TW','HK','AU','NZ','CA','MX','BR','AR','CL','AE','SA','IN','ZA'];

       export async function shouldShowOnboarding({ user, db } = {}) {
         const completed = localStorage.getItem('lumi_onboarding_completed') === 'true';
         if (completed) return false;
         const seen = localStorage.getItem('lumi_onboarding_seen') === 'true';
         if (seen) return false;
         if (user?.uid && db) {
           try {
             const snap = await db.collection('users').doc(user.uid).get();
             const prefs = snap.exists ? snap.data()?.preferences : null;
             if (prefs?.version >= 1) {
               localStorage.setItem('lumi_onboarding', JSON.stringify({ ...prefs, completedAt: prefs.completedAt || Date.now() }));
               localStorage.setItem('lumi_onboarding_seen', 'true');
               localStorage.setItem('lumi_onboarding_completed', 'true');
               setLocale({ lang: prefs.lang, country: prefs.country });
               return false;
             }
           } catch {}
         }
         return true;
       }

       export function startOnboarding(options = {}) {
         localStorage.setItem('lumi_onboarding_seen', 'true');
         // render DOM, manage step state, call completeStep on user submit per step
       }

       export function completeStep(stepNum, data, options = {}) {
         const current = JSON.parse(localStorage.getItem('lumi_onboarding') || '{}');
         const next = { ...current, ...data };
         localStorage.setItem('lumi_onboarding', JSON.stringify(next));
         if (stepNum === 1 && data.lang) setLocale({ lang: data.lang });
         if (stepNum === 2 && data.country) setLocale({ country: data.country });
         if (stepNum === 3) {
           next.completedAt = Date.now();
           localStorage.setItem('lumi_onboarding', JSON.stringify(next));
           localStorage.setItem('lumi_onboarding_completed', 'true');
           if (options.user && options.db) {
             options.db.collection('users').doc(options.user.uid).set({ preferences: { ...next, version: 1 } }, { merge: true }).catch(console.error);
           }
         }
       }

       export function skipOnboarding(options = {}) {
         const locale = getLocale();
         const defaults = { lang: locale.lang, country: locale.country, ownedPlatforms: [], completedAt: Date.now(), skipped: ['step1','step2','step3'] };
         localStorage.setItem('lumi_onboarding', JSON.stringify(defaults));
         localStorage.setItem('lumi_onboarding_seen', 'true');
         localStorage.setItem('lumi_onboarding_completed', 'true');
         if (options.user && options.db) {
           options.db.collection('users').doc(options.user.uid).set({ preferences: { ...defaults, version: 1 } }, { merge: true }).catch(console.error);
         }
       }

       export async function fetchProviders(country) {
         try {
           const res = await fetch(`/api/tmdb?path=/watch/providers/tv&watch_region=${country}`);
           if (!res.ok) return [];
           const data = await res.json();
           return (data.results || []).sort((a,b)=>a.display_priority-b.display_priority).slice(0, 20);
         } catch { return []; }
       }
       ```
