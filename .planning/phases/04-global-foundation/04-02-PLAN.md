    1. Create `src/lib/locale.js`:
       ```js
       const LANG_TO_COUNTRY = { tr:'TR', en:'US', de:'DE', fr:'FR', es:'ES', ja:'JP', ko:'KR', zh:'CN' };
       const COUNTRY_TO_LANG = { TR:'tr', US:'en', GB:'en', DE:'de', AT:'de', CH:'de', FR:'fr', ES:'es', MX:'es', AR:'es', JP:'ja', KR:'ko', CN:'zh', TW:'zh', HK:'zh' };
       const SUPPORTED_LANGS = ['tr','en','de','fr','es','ja','ko','zh'];

       export function getLocale() { return resolveLocaleSync(); }

       export function resolveLocaleSync() {
         // 1) stored
         try {
           const stored = JSON.parse(localStorage.getItem('lumi_locale') || 'null');
           if (stored?.lang && stored?.country) return { ...stored, source: 'stored' };
         } catch {}
         // 2) legacy appLanguage migration
         const legacy = localStorage.getItem('appLanguage');
         if (legacy && SUPPORTED_LANGS.includes(legacy)) {
           return { lang: legacy, country: LANG_TO_COUNTRY[legacy] || 'US', source: 'migrated' };
         }
         // 3) navigator.language via Intl.Locale
         const nav = (typeof navigator !== 'undefined' && navigator.language) || null;
         if (nav) {
           const loc = new Intl.Locale(nav);
           const lang = SUPPORTED_LANGS.includes(loc.language) ? loc.language : 'en';
           const country = loc.region || LANG_TO_COUNTRY[lang] || 'US';
           return { lang, country, source: 'navigator' };
         }
         // 4) default
         return { lang: 'en', country: 'US', source: 'default' };
       }

       export async function resolveLocaleAsync() {
         const sync = resolveLocaleSync();
         if (sync.source === 'stored' || sync.source === 'migrated') return sync;
         // refine country via geoip
         try {
           const res = await fetch('/api/geoip');
           const { countryCode } = await res.json();
           if (countryCode && countryCode !== 'XX') {
             const lang = sync.source === 'navigator' ? sync.lang : (COUNTRY_TO_LANG[countryCode] || 'en');
             return { lang, country: countryCode, source: 'geoip' };
           }
         } catch {}
         return sync;
       }

       export function setLocale(partial) {
         const current = resolveLocaleSync();
         const next = { ...current, ...partial, source: 'user' };
         localStorage.setItem('lumi_locale', JSON.stringify({ lang: next.lang, country: next.country, source: 'user' }));
         localStorage.setItem('appLanguage', next.lang); // legacy write-through
         if (typeof window !== 'undefined' && window.i18n) window.i18n.setLanguage(next.lang);
         return next;
       }

       export function parseAcceptLanguage(header) {
         if (!header) return [];
         return header.split(',').map(s => {
           const [tag, q] = s.trim().split(';q=');
           return { tag, q: parseFloat(q ?? '1') };
         }).sort((a, b) => b.q - a.q);
       }

       export { LANG_TO_COUNTRY, COUNTRY_TO_LANG, SUPPORTED_LANGS };
       ```
