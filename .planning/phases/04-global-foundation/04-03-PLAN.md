    2. Create `src/features/auth-migration.js`:
       ```js
       export const MIGRATION_FLAG = 'lumi_migrated_v1';

       export async function migrateOnAuth(user, db, options = {}) {
         if (localStorage.getItem(MIGRATION_FLAG) === 'true') return { skipped: true };
         if (!user?.uid) return { skipped: true, reason: 'no-user' };

         const liked = JSON.parse(localStorage.getItem('liked_items') || '[]');
         const watch = JSON.parse(localStorage.getItem('watchlist_items') || '[]');
         if (liked.length === 0 && watch.length === 0) {
           localStorage.setItem(MIGRATION_FLAG, 'true');
           return { migrated: 0 };
         }

         try {
           const batch = db.batch();
           for (const item of liked) batch.set(db.collection(`users/${user.uid}/favorites`).doc(String(item.id)), item, { merge: true });
           for (const item of watch) batch.set(db.collection(`users/${user.uid}/watchlist`).doc(String(item.id)), item, { merge: true });
           await batch.commit();
           localStorage.setItem(MIGRATION_FLAG, 'true');
           return { migrated: liked.length + watch.length };
         } catch (err) {
           console.error('[migration] failed', err);
           // flag NOT set → retries on next auth event
           if (options.onError) options.onError(err);
           return { error: err.message };
         }
       }
       ```

       ```js
       let activeResolve = null;
       let activeReject = null;

       export async function requireAuth({ action }) {
         const auth = window.firebase?.auth?.();
         if (auth?.currentUser) return auth.currentUser;
         return new Promise((resolve, reject) => {
           activeResolve = resolve;
           activeReject = reject;
           openAuthModal(action);
         });
       }

       export function openAuthModal(actionKey) {
         const modal = document.getElementById('auth-modal') || createModalDOM();
         // i18n: t(`authGate.${actionKey}`) for context line; fallback t('authGate.default')
         const ctxLine = window.i18n.t(`authGate.${actionKey}`) || window.i18n.t('authGate.default');
         modal.querySelector('.auth-modal-context').textContent = ctxLine;
         modal.classList.add('open');

         // Bind sign-in button click → calls public/services/auth.js → onAuthStateChanged fires elsewhere → activeResolve(user)
         // Bind cancel → activeReject(new DOMException('cancelled','AbortError')); closeAuthModal()
       }

       export function closeAuthModal() { /* hide modal */ }

       // Called by src/main.js auth listener
       export function _onAuthResolved(user) {
         if (activeResolve) { activeResolve(user); activeResolve = null; activeReject = null; closeAuthModal(); }
       }
       ```
